import { createHash } from "node:crypto";
import type { LanguageModelV3StreamPart } from "@ai-sdk/provider";
import { MockLanguageModelV3 } from "ai/test";
import { afterEach, describe, expect, it } from "vitest";
import { runNormalChatCompletion } from "$lib/server/services/normal-chat-model/completion";
import type { NormalChatOutputEvent } from "$lib/shared/normal-chat-output";
import { accountingReadFixture } from "./accounting-read-fixture";

// Review measurement (not a product test): what one conversation costs to read
// once its replies carry the per-token `normalChatOutput.events` that
// finalize.ts persists. Uses the real engine to make the events, the real
// migrated SQLite schema, and the real message readers.

const usage = {
	inputTokens: { total: 1, noCache: 1, cacheRead: 0, cacheWrite: 0 },
	outputTokens: { total: 1, text: 1, reasoning: 0 },
};
const provider = {
	id: "native",
	name: "native",
	displayName: "Native",
	baseUrl: "https://native.invalid",
	modelName: "native",
	apiKey: "synthetic",
	maxOutputTokens: 1000,
	maxModelContext: 32000,
};

async function engineEvents(
	requestId: string,
	reasoningTokens: number,
	answerTokens: number,
): Promise<NormalChatOutputEvent[]> {
	const segmentId = createHash("sha256")
		.update(JSON.stringify(["u", "c", requestId]))
		.digest("hex");
	const parts: LanguageModelV3StreamPart[] = [
		{ type: "reasoning-start", id: "reasoning-0" },
	];
	for (let i = 0; i < reasoningTokens; i++)
		parts.push({ type: "reasoning-delta", id: "reasoning-0", delta: " word" });
	parts.push({ type: "reasoning-end", id: "reasoning-0" });
	parts.push({ type: "text-start", id: "txt-0" });
	for (let i = 0; i < answerTokens; i++)
		parts.push({ type: "text-delta", id: "txt-0", delta: " word" });
	parts.push({ type: "text-end", id: "txt-0" });
	parts.push({
		type: "finish",
		usage,
		finishReason: { unified: "stop", raw: "stop" },
	});
	const model = new MockLanguageModelV3({
		doStream: async () => ({
			stream: new ReadableStream<LanguageModelV3StreamPart>({
				start(controller) {
					for (const part of parts) controller.enqueue(part);
					controller.close();
				},
			}),
		}),
	});
	const run = runNormalChatCompletion(
		{
			provider,
			messages: [{ role: "user" as const, content: "synthetic request" }],
			completionPolicy: {
				turnId: segmentId,
				segmentId,
				deadlineAt: Date.now() + 300000,
				totalTimeoutMs: 300000,
				forAttempt: () => ({ classifyExecution: () => "succeeded" as const }),
			},
		},
		{
			model,
			maxModelContext: 32000,
			providerOptions: undefined,
			stopWhen: undefined,
			metadata: (responseModelName: string) => ({
				modelId: "native",
				providerId: "native",
				providerName: "native",
				displayName: "Native",
				requestedModelName: "native",
				responseModelName,
			}),
			mapUsage: () => ({ inputTokens: 1, outputTokens: 1, totalTokens: 2 }),
		},
		true,
	);
	for (;;) {
		const next = await run.next();
		if (next.done) {
			const snapshot = next.value.continuation;
			if (!snapshot) throw new Error("no continuation");
			return snapshot.events.filter((e) => e.segmentId === snapshot.segmentId);
		}
	}
}

let fixture: Awaited<ReturnType<typeof accountingReadFixture>> | undefined;
afterEach(() => {
	fixture?.close();
	fixture = undefined;
});

function timed<T>(run: () => Promise<T>) {
	const started = performance.now();
	return run().then((value) => ({ value, ms: performance.now() - started }));
}

describe("review measurement: read cost of persisted structural events", () => {
	it("compares a 50-reply conversation with and without persisted events", async () => {
		// 1,200 reasoning + 1,957 answer deltas: the shape of the one real Kimi
		// turn recorded by the Natural v10 staging probe (4,417 frames, 2.49 MB).
		const events = await engineEvents("stream", 1200, 1957);
		const turns = 50;
		const rows: Record<string, unknown>[] = [];
		for (const withEvents of [false, true]) {
			fixture = await accountingReadFixture();
			for (let i = 0; i < turns; i++) {
				fixture.message(`u${i}`, "user");
				fixture.message(
					`a${i}`,
					"assistant",
					withEvents ? { normalChatOutput: { version: 1, events } } : {},
				);
			}
			const dbBytes = Number(
				(
					fixture.sqlite
						.prepare(
							"select sum(length(metadata_json)) as bytes from messages where conversation_id = 'c'",
						)
						.get() as { bytes: number }
				).bytes,
			);
			const window = await timed(() =>
				fixture!.readers.listMessageWindow("c", {}),
			);
			const all = await timed(() => fixture!.readers.listMessages("c"));
			const payloadStarted = performance.now();
			const payload = JSON.stringify(window.value.messages);
			const serializeMs = performance.now() - payloadStarted;
			rows.push({
				withEvents,
				messagesInWindow: window.value.messages.length,
				storedMetadataBytes: dbBytes,
				listMessageWindowMs: Math.round(window.ms),
				listMessagesMs: Math.round(all.ms),
				detailPayloadBytes: Buffer.byteLength(payload),
				serializePayloadMs: Math.round(serializeMs),
			});
			fixture.close();
			fixture = undefined;
		}
		console.log(JSON.stringify(rows, null, 1));
		expect(rows).toHaveLength(2);
	}, 300000);
});
