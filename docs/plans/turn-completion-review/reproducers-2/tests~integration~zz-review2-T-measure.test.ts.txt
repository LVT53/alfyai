function stream(parts: LanguageModelV3StreamPart[]) {
	return new ReadableStream<LanguageModelV3StreamPart>({
		start(controller) {
			for (const part of parts) controller.enqueue(part);
			controller.close();
		},
	});
}
const sid = (request: string) =>
	createHash("sha256")
		.update(JSON.stringify(["user", "conversation", request]))
		.digest("hex");

async function drive(
	calls: LanguageModelV3StreamPart[][],
	request = "stream",
	extra: Record<string, unknown> = {},
) {
	const segmentId = sid(request);
	let n = 0;
	const model = new MockLanguageModelV3({
		doStream: async () => ({ stream: stream(calls[Math.min(n++, calls.length - 1)]) }),
	});
	const wire: NormalChatOutputEvent[] = [];
	const gen = runNormalChatCompletion(
		{
			provider,
			messages: [{ role: "user" as const, content: "synthetic request" }],
			completionPolicy: {
				turnId: segmentId,
				segmentId,
				deadlineAt: Date.now() + 300000,
				totalTimeoutMs: 300000,
				forAttempt: () => ({ classifyExecution: () => "succeeded" as const }),
				onOutputEvent: (event: NormalChatOutputEvent) =>
					wire.push(JSON.parse(JSON.stringify(event))),
			},
			...extra,
		} as never,
		{
			model,
			maxModelContext: 32000,
			providerOptions: undefined,
			stopWhen: undefined,
			metadata: (responseModelName: string) => ({
				modelId: "native",
				providerId: "native",
				providerName: "native",
				displayName: "Native",
				requestedModelName: "native",
				responseModelName,
			}),
			mapUsage: () => ({ inputTokens: 1, outputTokens: 1, totalTokens: 2 }),
		},
		true,
	);
	let result: Awaited<ReturnType<typeof gen.next>>["value"];
	for (;;) {
		const next = await gen.next();
		if (next.done) {
			result = next.value;
			break;
		}
	}
	const snapshot = (result as { continuation?: { events: NormalChatOutputEvent[]; segmentId: string } }).continuation;
	if (!snapshot) throw new Error("no continuation");
	const seg = (e: NormalChatOutputEvent) => e.segmentId === snapshot.segmentId;
	return {
		calls: n,
		wire,
		full: wire.filter(seg),
		// finalize.ts:592-598 exactly
		stored: compactNormalChatOutputEvents(snapshot.events.filter(seg)),
		result,
	};
}

function answerParts(reasoning: number, answer: number, token = " word") {
	const parts: LanguageModelV3StreamPart[] = [];
	if (reasoning) {
		parts.push({ type: "reasoning-start", id: "r" });
		for (let i = 0; i < reasoning; i++)
			parts.push({ type: "reasoning-delta", id: "r", delta: token });
		parts.push({ type: "reasoning-end", id: "r" });
	}
	parts.push({ type: "text-start", id: "t" });
	for (let i = 0; i < answer; i++)
		parts.push({ type: "text-delta", id: "t", delta: token });
	parts.push({ type: "text-end", id: "t" }, finish());
	return parts;
}

function view(events: readonly NormalChatOutputEvent[]) {
	const p = projectNormalChatOutput(events);
	return {
		answer: p.answer,
		activity: p.activity.map((b) => [b.stepId, b.role, b.text]),
		blocks: p.blocks.map((b) => [b.stepId, b.kind, b.role, b.text, b.ended]),
		completion: p.completion,
	};
}

describe("B: wire bytes per token", () => {
	it("compact vs canonical at realistic cadence, decoded equality", async () => {
		const run = await drive([answerParts(1200, 1957)]);
		const deltas = run.wire.filter((e) => e.type === "block_delta").length;
		const rows = [];
		for (const gapMs of [5, 20, 50, 100]) {
			for (const compact of [true, false]) {
				vi.useFakeTimers();
				const frames: string[] = [];
				const writer = createOutputEventWriter(
					(frame) => {
						frames.push(frame);
						return true;
					},
					{ compact },
				);
				for (const event of run.wire) {
					if (event.type === "block_delta") vi.advanceTimersByTime(gapMs);
					writer.write(event);
				}
				writer.flush();
				vi.useRealTimers();
				const bytes = frames.reduce((n, f) => n + Buffer.byteLength(f), 0);
				// decode as the browser does
				const decoder = createNormalChatWireDecoder();
				const acc = createNormalChatOutputAccumulator();
				const { frames: parsed } = consumeAiSdkUiStreamFrames(frames.join(""));
				for (const f of parsed)
					if (f.kind !== "done") {
						const e = decoder.consume(f.part);
						if (e) acc.apply(e);
					}
				const decoded = acc.getProjection();
				expect(decoded.answer).toBe(projectNormalChatOutput(run.full).answer);
				// relay for reconnect listeners
				const relay = createCanonicalOutputRelay();
				const relayed = frames.map((f) => relay(f)).join("");
				rows.push({
					gapMs,
					compact,
					frames: frames.length,
					bytes,
					bytesPerToken: +(bytes / deltas).toFixed(1),
					relayBytes: Buffer.byteLength(relayed),
				});
			}
		}
		const naive = run.wire.reduce(
			(n, e) => n + Buffer.byteLength(streamNormalChatOutputEvent(e)),
			0,
		);
		process.stdout.write(
			`B-wire ${JSON.stringify({ deltas, naivePerTokenBytes: +(naive / deltas).toFixed(1), rows })}\n`,
		);
	}, 120000);
});

describe("C: replay bound", () => {
	async function overflow(reasoning: number, answer: number, token: string, gapMs = 20) {
		const run = await drive([answerParts(reasoning, answer, token)], `c-${token.length}-${answer}`);
		vi.useFakeTimers();
		const journal = new StreamReplayJournal();
		const relay = createCanonicalOutputRelay();
		let tokens = 0;
		let at: number | null = null;
		const writer = createOutputEventWriter(
			(frame) => {
				journal.append(relay(frame));
				if (at === null && !journal.read().complete) at = tokens;
				return true;
			},
			{ compact: true },
		);
		for (const event of run.wire) {
			if (event.type === "block_delta") {
				tokens++;
				vi.advanceTimersByTime(gapMs);
			}
			writer.write(event);
		}
		writer.flush();
		vi.useRealTimers();
		const read = journal.read();
		return {
			reasoning,
			answer,
			tokenBytes: Buffer.byteLength(JSON.stringify(token)) - 2,
			overflowAtToken: at,
			completeAtEnd: read.complete,
			retainedBytes: read.complete ? Buffer.byteLength(read.chunks.join("")) : 0,
		};
	}
	it("finds the token count where replay stops", async () => {
		const rows = [];
		rows.push(await overflow(0, 45000, " word"));
		rows.push(await overflow(20000, 25000, " word"));
		rows.push(await overflow(0, 30000, " szó,"));
		rows.push(await overflow(0, 30000, '\t"x\\'));
		process.stdout.write(`C-replay ${JSON.stringify(rows)}\n`);
	}, 300000);
});
